Problem statement

Imagine a classic catapult game, but set entirely in a messy playroom. You and your opponent build custom forts out of wooden blocks, plastic bricks, and dominoes. Once the forts are built, you take turns launching bouncy balls, toy cars, and action figures at each other's creations to knock them down. The winner is the player who has the last toy castle standing!

    it('shows loading then an accessible API error and successfully retries', { tags: ['frontend-list-failure'] }, async t => {
  const page = await pageFor(t);
  let finishRequest;
  const pending = new Promise(resolve => { finishRequest = resolve; });
  let requests = 0;
  await page.route('**/list-toys', async route => {
    requests++;
    if (requests === 1) {
      await pending;
      await route.fulfill({ status: 503, json: { error: 'Service unavailable' } });
    } else {
      await route.fulfill({ json: { toys: [{ id: 'retry-blocks', name: 'Recovered blocks' }] } });
    }
  });
  await page.goto(frontend.origin);
  try {
    await page.getByRole('status').filter({ hasText: 'Loading building toys' }).waitFor({ timeout: 5000 });
  } finally {
    finishRequest();
  }
  await page.getByRole('alert').waitFor({ timeout: 5000 });
  assert.equal(await page.getByRole('radio').count(), 0);
  await page.getByRole('button', { name: 'Try again', exact: true }).click();
  await page.getByRole('radio', { name: 'Recovered blocks', exact: true }).waitFor({ timeout: 5000 });
  assert.equal(requests, 2);
  assert.equal(await page.getByRole('alert').count(), 0);
})
